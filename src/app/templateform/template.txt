import { Component, OnInit } from '@angular/core';
import { NgForm } from '@angular/forms';
import { Employee, EmployeeService } from '../services/employee.service';

@Component({
  selector: 'app-templateform',
  templateUrl: './templateform.component.html',
  styleUrls: ['./templateform.component.css']
})
export class TemplateformComponent implements OnInit {

  employees: Employee[] = [];
  employee: Employee = this.getEmptyEmployee();
  editMode = false;
  customError = '';

  constructor(private empService: EmployeeService) {}

  ngOnInit() {
    this.employees = this.empService.getAll();
  }

  getEmptyEmployee(): Employee {
    return {
      id: 0,
      name: '',
      email: '',
      gender: '',
      department: '',
      joiningDate: '',
      salary: 0,
      isPermanent: false
    };
  }

  // Custom validation before save
  validateCustom(emp: Employee): boolean {
    if (emp.name.toLowerCase() === 'test') {
      this.customError = 'Name "test" is not allowed!';
      return false;
    }
    this.customError = '';
    return true;
  }

  onSubmit(form: NgForm) {
    if (!this.validateCustom(this.employee)) return;

    if (this.editMode) {
      this.empService.update(this.employee.id, this.employee);
      this.editMode = false;
    } else {
      this.empService.add(this.employee);
    }

    this.employees = this.empService.getAll();
    this.resetForm(form);
  }

  onEdit(emp: Employee) {
    this.employee = { ...emp };
    this.editMode = true;
  }

  onDelete(id: number) {
    this.empService.delete(id);
    this.employees = this.empService.getAll();
  }

  resetForm(form: NgForm) {
    form.resetForm();
    this.employee = this.getEmptyEmployee();
    this.editMode = false;
    this.customError = '';
  }
}
